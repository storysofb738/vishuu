<script>
    // Particle Background Logic
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    let particles = [];

    function resize() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
    }
    window.addEventListener('resize', resize);
    resize();

    class Particle {
        constructor() {
            this.x = Math.random() * canvas.width;
            this.y = Math.random() * canvas.height;
            this.size = Math.random() * 1.5;
            this.speedX = Math.random() * 0.5 - 0.25;
            this.speedY = Math.random() * 0.5 - 0.25;
        }
        update() {
            this.x += this.speedX;
            this.y += this.speedY;
            if (this.x > canvas.width) this.x = 0;
            if (this.x < 0) this.x = canvas.width;
            if (this.y > canvas.height) this.y = 0;
            if (this.y < 0) this.y = canvas.height;
        }
        draw() {
            ctx.fillStyle = 'rgba(0, 242, 255, 0.3)';
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    for (let i = 0; i < 100; i++) particles.push(new Particle());

    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => { p.update(); p.draw(); });
        requestAnimationFrame(animate);
    }
    animate();

    // Typing Logic
    const message = [
        "You must be missing my “hiiii” voice... 🎙️",
        "I called you but you were busy, so I thought let me wish you differently this time.",
        "Btw, I remember your birthday... see? ✨",
        "But since I am out of town, this is my personal robot keeping the tradition alive. 🤖",
        "I will call you in the evening right after my flight takes off. ✈️",
        "Happy Birthday, Vishuuu! ❤️"
    ];

    let lineIndex = 0;
    let charIndex = 0;
    const output = document.getElementById('output');
    const startBtn = document.getElementById('startBtn');

    function initMessage() {
        startBtn.style.display = 'none';
        typeLine();
    }

    function typeLine() {
        if (lineIndex < message.length) {
            let currentLine = document.createElement('p');
            if(lineIndex === 2 || lineIndex === 5) currentLine.className = 'highlight';
            output.appendChild(currentLine);
            typeChar(currentLine, message[lineIndex]);
        }
    }

    function typeChar(element, text) {
        if (charIndex < text.length) {
            element.innerHTML += text.charAt(charIndex);
            charIndex++;
            setTimeout(() => typeChar(element, text), 40);
        } else {
            charIndex = 0;
            lineIndex++;
            setTimeout(typeLine, 600);
        }
    }
</script>
